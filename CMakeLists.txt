cmake_minimum_required(VERSION 3.21)
project(OcctQtTutorial VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(OpenCASCADE REQUIRED)
if(NOT OpenCASCADE_MAJOR_VERSION EQUAL 7 OR NOT OpenCASCADE_MINOR_VERSION EQUAL 9)
  message(FATAL_ERROR "Tutorial requires OCCT 7.9.x (tested: 7.9.3)")
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(cad_model modeling.cpp modeling.h)
target_include_directories(cad_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(cad_model PUBLIC TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep
  TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKFillet TKOffset TKMesh TKShHealing
  TKXSBase TKDESTEP TKDESTL)
qt_add_executable(OcctQtStarter main.cpp cad_view.h native_window.cpp native_window.h
  gui_experiments.cpp gui_experiments.h)
set_target_properties(OcctQtStarter PROPERTIES MACOSX_BUNDLE TRUE)
target_link_libraries(OcctQtStarter PRIVATE cad_model Qt6::Widgets Qt6::Test TKService TKV3d TKOpenGl)
if(APPLE)
  target_link_libraries(OcctQtStarter PRIVATE "-framework Cocoa" "-framework OpenGL")
endif()
if(MSVC)
  target_compile_options(cad_model PUBLIC /utf-8 /W4)
endif()
if(BUILD_TESTING)
  add_executable(model_tests tests/model_tests.cpp)
  target_link_libraries(model_tests PRIVATE cad_model)
  add_test(NAME modeling COMMAND model_tests ${CMAKE_CURRENT_BINARY_DIR}/test-artifacts)
  option(OCCT_ENABLE_GUI_TESTS "Enable real-window OpenGL experiments" OFF)
  if(OCCT_ENABLE_GUI_TESTS)
    add_test(NAME gui_experiments COMMAND OcctQtStarter --self-test
      --output ${CMAKE_CURRENT_BINARY_DIR}/gui-artifacts)
    set_tests_properties(gui_experiments PROPERTIES TIMEOUT 180)
  endif()
endif()
